% Page de faux-titre
\newcommand{\halftitlepage}[1]{%
    \thispagestyle{frontmatter}%
    \vspace*{\fill}%
    \begin{center}%
        {\fontsize{24}{28}\selectfont\bfseries #1}%
    \end{center}%
    \vspace*{\fill}%
    \clearpage%
}

% Page de titre
\newcommand{\maintitlepage}[3]{%
    \thispagestyle{frontmatter}%
    \vspace*{\fill}%
    \begin{center}%
        {\fontsize{32}{38}\selectfont\bfseries #1}\\[1cm]%
        {\fontsize{14}{18}\selectfont #2}\\[3cm]%
        {\fontsize{14}{18}\selectfont\itshape #3}%
    \end{center}%
    \vspace*{\fill}%
    \clearpage%
}

% Page de copyright/mentions légales
\newcommand{\copyrightpage}[1]{%
    \thispagestyle{frontmatter}%
    \vspace*{\fill}%
    {\small #1}%
    \clearpage%
}

% Page de séparation de lettre
\newcommand{\letterpage}[1]{%
    \clearpage%
    \thispagestyle{letterpage}%
    \renewcommand{\currentletter}{#1}%
    \vspace*{\fill}%
    \begin{center}%
        {\fontsize{80}{95}\selectfont\bfseries #1}%
    \end{center}%
    \vspace*{\fill}%
    \clearpage%
}
