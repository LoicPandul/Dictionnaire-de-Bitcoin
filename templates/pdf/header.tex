% Packages pour les mathématiques
\usepackage{amsmath}
\usepackage{amssymb}

% Mise en page
\usepackage{fancyhdr}
\usepackage{colortbl}

% Code et listings
\usepackage{listings}
\usepackage{tcolorbox}

% Couleurs
\definecolor{customgray}{RGB}{246, 248, 250}

% Configuration des listings
\lstdefinelanguage{text}{
    basicstyle=\ttfamily\small,
    morekeywords={}
}
\lstdefinelanguage{plaintext}{
    basicstyle=\ttfamily\small,
    morekeywords={}
}
\lstset{
    backgroundcolor=\color{customgray},
    frame=none,
    basicstyle=\ttfamily\small,
    aboveskip=0.5em,
    belowskip=0.5em,
    breaklines=true,
    breakatwhitespace=true
}

% Configuration tcolorbox
\tcbuselibrary{listingsutf8}
\tcbset{listing engine=listings}
\newtcblisting{codeblock}{
    colback=customgray,
    colframe=white,
    listing only,
    listing options={basicstyle=\ttfamily\small},
    left=1em,
    right=1em
}

% En-têtes et pieds de page
\pagestyle{fancy}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% Tableau sans lignes visibles
\arrayrulecolor{white}
